\section{Contenido}

\begin{frame}{Matemáticas}
  La transformada de Fourier en tiempo discreto de una secuencia \(x[n]\) es
  \begin{equation}
    X(e^{j\omega}) = \sum_{n=-\infty}^{\infty} x[n]\, e^{-j\omega n},
  \end{equation}
  y las ecuaciones de Maxwell en forma diferencial son
  \begin{align}
    \nabla \cdot \mathbf{E} &= \frac{\rho}{\varepsilon_0}, &
    \nabla \times \mathbf{E} &= -\frac{\partial \mathbf{B}}{\partial t}.
  \end{align}
  \begin{theorem}[Parseval]
    \(\sum_n |x[n]|^2 = \frac{1}{2\pi}\int_{-\pi}^{\pi} |X(e^{j\omega})|^2\,d\omega\).
  \end{theorem}
\end{frame}
